> **文档类型**：执行方案
> **基于**：诊断1-5 综合分析
> **目标**：将项目从"竞赛专用脚本"升级为"可维护的视觉处理库"
> **预计总工期**：Phase 1-4，约 60-80 小时

## 一、开篇判断

你的项目**功能完整、核心算法扎实**（原创的雷达点群/孤点算法、相机-雷达几何融合、ORB模板匹配），但处于"功能完成期"向"工程成熟期"过渡的节点。五份诊断对问题的判断高度一致，没有争议点。重构的核心目标应该是：

> **建立一套严格的数据流规范**：帧进来 → 检测器处理 → 结果输出，每个环节的输入输出格式严格定义。一旦这个数据流定义清楚，后续所有重构都是顺水推舟。

```python
from dataclasses import dataclass, field
from typing import Tuple, Optional, Dict, Any, List

@dataclass
class BoundingBox:
    x: int; y: int; width: int; height: int
    @property
    def center(self) -> Tuple[int, int]:
        return (self.x + self.width // 2, self.y + self.height // 2)
    @property
    def area(self) -> int:
```

```python
from abc import ABC, abstractmethod
import cv2, numpy as np
from src.core.types import DetectionResult

class ROIExtractor(ABC):
    @abstractmethod
    def extract(self, frame: np.ndarray, results: List[DetectionResult]) -> List[np.ndarray]: pass

class CircleROIExtractor(ROIExtractor):
    def __init__(self, radius_ratio=0.4):
        self.radius_ratio = radius_ratio
    def extract(self, frame, results):
        rois = []
        for r in results:
            cx, cy = r.center
            h, w = frame.shape[:2]
            radius = int(min(w, h) * self.radius_ratio)
            mask = np.zeros_like(frame[:,:,0])
            cv2.circle(mask, (cx, cy), radius, 255, -1)
            rois.append(cv2.bitwise_and(frame, frame, mask=mask))
        return rois

class RectROIExtractor(ROIExtractor): ...
class ORBROIExtractor(ROIExtractor): ...
```

**`src/pipeline/orchestrator.py`** — 将 `give_me_a_color_and_i_will_give_you_a_shape()` 改写为 Pipeline 编排器：

```python
class DetectionPipeline:
    def __init__(self, roi_extractor=None, shape_classifier=None):
        self.roi_extractor = roi_extractor or CircleROIExtractor()
        self.shape_classifier = shape_classifier or ShapeClassifier()

    def run(self, frame, color: str) -> List[DetectionResult]:
        color_results = detect_color_adapted(frame, color)
        if not color_results: return []
        rois = self.roi_extractor.extract(frame, color_results)
        final_results = []
        for roi, color_result in zip(rois, color_results):
            shapes = self.shape_classifier.classify(roi)
            for shape in shapes:
                shape.extra["color"] = color
                shape.extra["color_center"] = color_result.center
            final_results.extend(shapes)
        return final_results
```

```python
from abc import ABC, abstractmethod

class ModeHandler(ABC):
    def __init__(self, cap, serial, radar, pipeline):
        self.cap = cap; self.serial = serial
        self.radar = radar; self.pipeline = pipeline

    @abstractmethod
    async def process(self, frame): pass

    def _send_result(self, results):
        if results: self.serial.send(results)
```

#### T3.1 解耦 ROS 依赖

```python
# src/radar/base.py
from abc import ABC, abstractmethod
class RadarSource(ABC):
    @abstractmethod
    def get_scan(self) -> RadarScan: ...

# src/radar/ros_source.py
class ROSRadarSource(RadarSource):
    def __init__(self, topic="/scan"):
        import rospy
        from sensor_msgs.msg import LaserScan
        self._data = None
        rospy.Subscriber(topic, LaserScan, self._callback)
    def get_scan(self) -> RadarScan:
        return self._to_scan(self._data)

# src/radar/sim_source.py
class SimRadarSource(RadarSource):
    """用于无 ROS 环境的模拟数据源"""
    def __init__(self, data_file=None):
        self._data = self._load_or_generate(data_file)
        self._idx = 0
    def get_scan(self) -> RadarScan:
        result = self._data[self._idx]
        self._idx = (self._idx + 1) % len(self._data)
        return result

# src/radar/fusion.py
class RadarFusion:
    def __init__(self, source: RadarSource = None):
        self._source = source or self._auto_detect_source()
    def _auto_detect_source(self) -> RadarSource:
        try:
            return ROSRadarSource()
        except ImportError:
            return SimRadarSource()
```

```python
    frame_count = detection_count = 0
    while cap.isOpened():
        ret, frame = cap.read()
```

| 风险 | 概率 | 影响 | 应对策略 |

## 七、不在此计划范围内的建议

1. **异步重构**：当前 `asyncio` 和 `cv2.imshow()` 混用，建议后续评估是否用 `threading` 或 `multiprocessing` 替代
2. **性能优化**：各检测器有大量重复的 `cv2.cvtColor()` 调用，可提取到 Pipeline 入口做一次转换
3. **模型化**：考虑用 ONNX Runtime 部署轻量模型（如 YOLOv8）替代部分传统算法
4. **CI/CD**：使用 GitHub Actions 自动化测试，每次 push 时运行单元测试和集成测试
